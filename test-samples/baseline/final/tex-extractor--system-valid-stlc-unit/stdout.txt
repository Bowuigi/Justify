\newcommand{\jyGrammar}[0]{
  \begin{array}{rll}
    & & \textbf{Terms} \\
    e \mathrel{::=} & \jgTermVariable{x} & \text{Variable $x$} \\
    \mid & \jgTermLambda{x}{\tau}{e} & \text{Binds the applied term of type $t$ to the variable $x$ in $e$} \\
    \mid & \jgTermApply{e_1}{e_2} & \text{Applies $e_2$ to $e_1$, performing beta reduction} \\
    \mid & \jgTermStar{} & \text{The only term of type $\mathbf{1}$} \\
    \\[-5pt] & & \textbf{Types} \\
    \tau , A , B \mathrel{::=} & \jgTypeArrow{A}{B} & \text{A function from $A$ to $B$} \\
    \mid & \jgTypeUnit{} & \text{A type with only one inhabitant, $\star$} \\
    \\[-5pt] & & \textbf{Contexts} \\
    \Gamma \mathrel{::=} & \jgContextEmpty{} & \text{Empty context, empty environment} \\
    \mid & \jgContextExtend{\Gamma}{x}{\tau} & \text{A cons operation for contexts (here, assoc lists)}
  \end{array}}
